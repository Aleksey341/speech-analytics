#!/usr/bin/env python3
"""
Автоматизированный скрипт развертывания системы речевой аналитики
Выполняет полную установку, настройку и запуск системы
"""

import os
import sys
import subprocess
import json
import shutil
import platform
import argparse
from pathlib import Path
from typing import Dict, List, Optional
import requests
import time
import psutil

class Colors:
    """ANSI цвета для консольного вывода"""
    HEADER = '\033[95m'
    OKBLUE = '\033[94m'
    OKCYAN = '\033[96m'
    OKGREEN = '\033[92m'
    WARNING = '\033[93m'
    FAIL = '\033[91m'
    ENDC = '\033[0m'
    BOLD = '\033[1m'
    UNDERLINE = '\033[4m'

class SpeechAnalyticsDeployer:
    """Класс для автоматизированного развертывания системы"""
    
    def __init__(self, config_path: Optional[str] = None):
        self.config_path = config_path or "deployment_config.json"
        self.project_root = Path.cwd()
        self.venv_path = self.project_root / "venv"
        self.config = self.load_config()
        
    def load_config(self) -> Dict:
        """Загрузка конфигурации развертывания"""
        default_config = {
            "system": {
                "python_version": "3.9",
                "project_name": "speech-analytics",
                "environment": "development",  # development, production
                "port": 5000,
                "host": "0.0.0.0"
            },
            "database": {
                "type": "sqlite",  # sqlite, postgresql, mysql
                "name": "speech_analytics.db",
                "host": "localhost",
                "port": 5432,
                "username": "",
                "password": ""
            },
            "redis": {
                "enabled": True,
                "host": "localhost",
                "port": 6379,
                "password": ""
            },
            "nginx": {
                "enabled": True,
                "domain": "localhost",
                "ssl": False
            },
            "integrations": {
                "crm": {"enabled": False},
                "telephony": {"enabled": False},
                "messenger": {"enabled": False},
                "cloud_storage": {"enabled": False}
            },
            "monitoring": {
                "enabled": True,
                "log_level": "INFO"
            }
        }
        
        if os.path.exists(self.config_path):
            try:
                with open(self.config_path, 'r', encoding='utf-8') as f:
                    user_config = json.load(f)
                    # Объединяем с default_config
                    default_config.update(user_config)
            except Exception as e:
                self.print_error(f"Ошибка загрузки конфигурации: {e}")
                self.print_info("Используется конфигурация по умолчанию")
        
        return default_config
    
    def save_config(self):
        """Сохранение конфигурации"""
        with open(self.config_path, 'w', encoding='utf-8') as f:
            json.dump(self.config, f, indent=2, ensure_ascii=False)
    
    def print_header(self, text: str):
        """Вывод заголовка"""
        print(f"\n{Colors.HEADER}{Colors.BOLD}{'='*60}{Colors.ENDC}")
        print(f"{Colors.HEADER}{Colors.BOLD}{text.center(60)}{Colors.ENDC}")
        print(f"{Colors.HEADER}{Colors.BOLD}{'='*60}{Colors.ENDC}")
    
    def print_success(self, text: str):
        """Вывод сообщения об успехе"""
        print(f"{Colors.OKGREEN}✅ {text}{Colors.ENDC}")
    
    def print_info(self, text: str):
        """Вывод информационного сообщения"""
        print(f"{Colors.OKBLUE}ℹ️  {text}{Colors.ENDC}")
    
    def print_warning(self, text: str):
        """Вывод предупреждения"""
        print(f"{Colors.WARNING}⚠️  {text}{Colors.ENDC}")
    
    def print_error(self, text: str):
        """Вывод ошибки"""
        print(f"{Colors.FAIL}❌ {text}{Colors.ENDC}")
    
    def run_command(self, command: str, check: bool = True, shell: bool = True) -> subprocess.CompletedProcess:
        """Выполнение команды в терминале"""
        self.print_info(f"Выполняем: {command}")
        try:
            result = subprocess.run(
                command, 
                shell=shell, 
                check=check, 
                capture_output=True, 
                text=True
            )
            if result.stdout:
                print(result.stdout)
            return result
        except subprocess.CalledProcessError as e:
            self.print_error(f"Ошибка выполнения команды: {e}")
            if e.stderr:
                print(e.stderr)
            raise
    
    def check_system_requirements(self):
        """Проверка системных требований"""
        self.print_header("ПРОВЕРКА СИСТЕМНЫХ ТРЕБОВАНИЙ")
        
        # Проверка операционной системы
        os_name = platform.system()
        self.print_info(f"Операционная система: {os_name} {platform.release()}")
        
        if os_name not in ['Linux', 'Darwin', 'Windows']:
            self.print_warning(f"Неподдерживаемая ОС: {os_name}")
        
        # Проверка Python
        python_version = f"{sys.version_info.major}.{sys.version_info.minor}"
        self.print_info(f"Версия Python: {python_version}")
        
        required_version = self.config['system']['python_version']
        if python_version < required_version:
            self.print_error(f"Требуется Python {required_version} или выше")
            sys.exit(1)
        
        # Проверка доступной памяти
        memory = psutil.virtual_memory()
        memory_gb = memory.total / (1024**3)
        self.print_info(f"Доступная память: {memory_gb:.1f} GB")
        
        if memory_gb < 2:
            self.print_warning("Рекомендуется минимум 2 GB RAM")
        
        # Проверка свободного места на диске
        disk = psutil.disk_usage('/')
        free_gb = disk.free / (1024**3)
        self.print_info(f"Свободное место: {free_gb:.1f} GB")
        
        if free_gb < 5:
            self.print_warning("Рекомендуется минимум 5 GB свободного места")
        
        self.print_success("Проверка системных требований завершена")
    
    def setup_python_environment(self):
        """Настройка Python окружения"""
        self.print_header("НАСТРОЙКА PYTHON ОКРУЖЕНИЯ")
        
        # Создание виртуального окружения
        if not self.venv_path.exists():
            self.print_info("Создание виртуального окружения...")
            self.run_command(f"python -m venv {self.venv_path}")
            self.print_success("Виртуальное окружение создано")
        else:
            self.print_info("Виртуальное окружение уже существует")
        
        # Активация виртуального окружения и установка зависимостей
        if platform.system() == "Windows":
            pip_path = self.venv_path / "Scripts" / "pip"
            python_path = self.venv_path / "Scripts" / "python"
        else:
            pip_path = self.venv_path / "bin" / "pip"
            python_path = self.venv_path / "bin" / "python"
        
        # Обновление pip
        self.print_info("Обновление pip...")
        self.run_command(f"{python_path} -m pip install --upgrade pip")
        
        # Создание requirements.txt если не существует
        requirements_path = self.project_root / "requirements.txt"
        if not requirements_path.exists():
            self.create_requirements_file()
        
        # Установка зависимостей
        self.print_info("Установка зависимостей Python...")
        self.run_command(f"{pip_path} install -r requirements.txt")
        
        # Дополнительные зависимости для production
        if self.config['system']['environment'] == 'production':
            self.print_info("Установка production зависимостей...")
            self.run_command(f"{pip_path} install gunicorn supervisor")
        
        self.print_success("Python окружение настроено")

    def setup_database(self):
        """Настройка базы данных"""
        self.print_header("НАСТРОЙКА БАЗЫ ДАННЫХ")
        
        db_config = self.config['database']
        db_type = db_config['type']
        
        if db_type == 'sqlite':
            self.setup_sqlite()
        elif db_type == 'postgresql':
            self.setup_postgresql()
        elif db_type == 'mysql':
            self.setup_mysql()
        
        self.print_success("База данных настроена")
    
    def setup_sqlite(self):
        """Настройка SQLite"""
        self.print_info("Используется SQLite база данных")
        db_path = self.project_root / self.config['database']['name']
        
        # SQLite не требует дополнительной настройки
        self.print_info(f"База данных будет создана по пути: {db_path}")
    
    def setup_postgresql(self):
        """Настройка PostgreSQL"""
        self.print_info("Настройка PostgreSQL...")
        
        db_config = self.config['database']
        
        # Проверка доступности PostgreSQL
        try:
            import psycopg2
            conn = psycopg2.connect(
                host=db_config['host'],
                port=db_config['port'],
                user=db_config['username'],
                password=db_config['password'],
                database='postgres'  # подключаемся к системной БД
            )
            
            # Создание базы данных если не существует
            conn.autocommit = True
            cursor = conn.cursor()
            cursor.execute(f"SELECT 1 FROM pg_database WHERE datname = '{db_config['name']}'")
            
            if not cursor.fetchone():
                cursor.execute(f"CREATE DATABASE {db_config['name']}")
                self.print_success(f"База данных {db_config['name']} создана")
            else:
                self.print_info(f"База данных {db_config['name']} уже существует")
            
            cursor.close()
            conn.close()
            
        except ImportError:
            self.print_error("psycopg2 не установлен")
        except Exception as e:
            self.print_error(f"Ошибка подключения к PostgreSQL: {e}")

    def get_database_url(self) -> str:
        """Генерация URL базы данных"""
        db_config = self.config['database']
        
        if db_config['type'] == 'sqlite':
            return f"sqlite:///{db_config['name']}"
        elif db_config['type'] == 'postgresql':
            return f"postgresql://{db_config['username']}:{db_config['password']}@{db_config['host']}:{db_config['port']}/{db_config['name']}"
        elif db_config['type'] == 'mysql':
            return f"mysql://{db_config['username']}:{db_config['password']}@{db_config['host']}:{db_config['port']}/{db_config['name']}"
        
        return ""
    
    def setup_nginx(self):
        """Настройка Nginx"""
        if not self.config['nginx']['enabled']:
            return
        
        self.print_header("НАСТРОЙКА NGINX")
        
        nginx_config = f"""
server {{
    listen 80;
    server_name {self.config['nginx']['domain']};
    
    client_max_body_size 50M;
    
    location / {{
        proxy_pass http://127.0.0.1:{self.config['system']['port']};
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        proxy_connect_timeout 300s;
        proxy_send_timeout 300s;
        proxy_read_timeout 300s;
    }}
    
    location /static {{
        alias {self.project_root}/app/static;
        expires 30d;
    }}
}}
        """.strip()
        
        # Сохраняем конфигурацию
        nginx_config_path = self.project_root / "config" / "nginx.conf"
        with open(nginx_config_path, "w") as f:
            f.write(nginx_config)
        
        self.print_info(f"Конфигурация Nginx сохранена в {nginx_config_path}")
        self.print_warning("Не забудьте скопировать конфигурацию в /etc/nginx/sites-available/")

        service_content = f"""
[Unit]
Description=Speech Analytics Web Application
After=network.target

[Service]
Type=exec
User=www-data
Group=www-data
WorkingDirectory={self.project_root}
Environment=PATH={self.venv_path}/bin
ExecStart={self.venv_path}/bin/gunicorn --workers 4 --bind {self.config['system']['host']}:{self.config['system']['port']} app:app

    def test_installation(self):
        """Тестирование установки"""
        self.print_header("ТЕСТИРОВАНИЕ УСТАНОВКИ")
        
        # Запуск сервера в фоновом режиме для тестирования
        if platform.system() == "Windows":
            python_path = self.venv_path / "Scripts" / "python"
        else:
            python_path = self.venv_path / "bin" / "python"
        
        # Создаем простой тестовый сервер
        test_server_script = """
from flask import Flask, jsonify
app = Flask(__name__)

@app.route('/health')
def health():
    return jsonify({"status": "healthy", "message": "Speech Analytics API is running"})

if __name__ == '__main__':
    app.run(host='127.0.0.1', port=5555, debug=False)
        """
        
        with open("test_server.py", "w") as f:
            f.write(test_server_script)
        
        # Запускаем тестовый сервер
        try:
            self.print_info("Запуск тестового сервера...")
            process = subprocess.Popen([str(python_path), "test_server.py"])
            
            # Ждем запуска сервера
            time.sleep(3)
            
            # Тестируем доступность
            try:
                response = requests.get("http://127.0.0.1:5555/health", timeout=5)
                if response.status_code == 200:
                    self.print_success("Тестовый сервер отвечает")
                    data = response.json()
                    self.print_info(f"Ответ сервера: {data['message']}")
                else:
                    self.print_warning(f"Сервер вернул код {response.status_code}")
            except requests.exceptions.RequestException as e:
                self.print_warning(f"Ошибка подключения к серверу: {e}")
            
            # Останавливаем тестовый сервер
            process.terminate()
            process.wait()
            
        except Exception as e:
            self.print_error(f"Ошибка тестирования: {e}")
        finally:
            if os.path.exists("test_server.py"):
                os.remove("test_server.py")
        
        self.print_success("Тестирование завершено")

        self.print_success(f"Скрипт запуска создан: {script_path}")

def main():
    """Главная функция"""
    parser = argparse.ArgumentParser(description="Speech Analytics Deployment Tool")
    parser.add_argument("--config", help="Путь к файлу конфигурации")
    parser.add_argument("--step", help="Выполнить только определенный шаг", 
                       choices=['requirements', 'python', 'database', 'redis', 'nginx', 'test'])
    parser.add_argument("--env", help="Окружение развертывания", 
                       choices=['development', 'production'], default='development')
    
    args = parser.parse_args()
    
    # Создаем экземпляр деплойера
    deployer = SpeechAnalyticsDeployer(args.config)
    
    # Устанавливаем окружение
    deployer.config['system']['environment'] = args.env
    
    if args.step:
        # Выполняем только определенный шаг
        step_methods = {
            'requirements': deployer.check_system_requirements,
            'python': deployer.setup_python_environment,
            'database': deployer.setup_database,
            'redis': deployer.setup_redis,
            'nginx': deployer.setup_nginx,
            'test': deployer.test_installation
        }
        
        if args.step in step_methods:
            step_methods[args.step]()
        else:
            deployer.print_error(f"Неизвестный шаг: {args.step}")
    else:
        # Полное развертывание
        deployer.full_deployment()

if __name__ == "__main__":
    main()
